    <script>
        const baseCategories = [
            { id: 1, name: '1. Rice', list: 'riceList', def: 'Plain Rice' },
            { id: 2, name: '2. Roti', list: 'rotiList', def: 'Chapati' },
            { id: 3, name: '3. Dal', list: 'dalList', def: 'Dal Fry' },
            { id: 4, name: '4. Dry Sabji', list: 'dryList', def: 'Mix Veg' },
            { id: 5, name: '5. Gravy Sabji', list: 'gravyList', def: 'Masoor Masala' }
        ];

        window.onload = () => {
            document.getElementById('startDate').valueAsDate = new Date();
            generateMenu();
        };
        
        document.getElementById('startDate').addEventListener('change', generateMenu);

        function generateMenu() {
            const container = document.getElementById('menuContainer');
            container.innerHTML = '';
            const client = document.getElementById('clientSelect').value;
            document.getElementById('clientDisplay').innerText = `CLIENT: ${client}`;

            const isHUL_UB = (client === "Hindustan Unilever Limited" || client === "United Breweries Limited");
            const daysToDisplay = isHUL_UB ? 7 : 6;

            let finalCategories = [...baseCategories];
            if (!isHUL_UB) {
                finalCategories.push({ id: 6, name: '6. Salad', list: 'standardSides', def: 'Salad' });
                finalCategories.push({ id: 7, name: '7. Pickle', list: 'standardSides', def: 'Pickle' });
                finalCategories.push({ id: 8, name: '8. Papad', list: 'standardSides', def: 'Papad' });
            } else {
                finalCategories.push({ id: 6, name: '6. Chicken/Fish/Egg', list: 'nonVegList', def: 'Chicken Masala', isRed: true });
                finalCategories.push({ id: 7, name: '7. Curd', list: 'curdList', def: 'Dahi' });
                finalCategories.push({ id: 8, name: '8. Paneer', list: 'paneerList', def: 'Paneer Masala' });
                finalCategories.push({ id: 9, name: '9. Sweets', list: 'sweetList', def: 'Gulab Jamun' });
            }

            let startDate = new Date(document.getElementById('startDate').value);
            let currentDate = new Date(startDate);
            let rangeStart = ""; let rangeEnd = "";

            for (let week = 1; week <= 2; week++) {
                const weekDiv = document.createElement('div');
                weekDiv.className = 'week-title';
                weekDiv.innerText = `WEEK ${week} SCHEDULE`;
                container.appendChild(weekDiv);

                const wrapper = document.createElement('div');
                wrapper.className = 'table-wrapper'; 

                const table = document.createElement('table');
                const thead = document.createElement('thead');
                const tbody = document.createElement('tbody');

                // Header Row Construction
                let headerRow = '<tr><th class="col-category">CATEGORY</th>';
                let colAdded = 0;
                while (colAdded < daysToDisplay) {
                    if (!isHUL_UB && currentDate.getDay() === 0) { currentDate.setDate(currentDate.getDate() + 1); continue; }
                    
                    // DATE FORMAT: DD-MM-YYYY
                    let dd = currentDate.getDate().toString().padStart(2, '0');
                    let mm = (currentDate.getMonth()+1).toString().padStart(2, '0');
                    let yyyy = currentDate.getFullYear();
                    let fullDateStr = `${dd}-${mm}-${yyyy}`;
                    
                    if (week === 1 && colAdded === 0) rangeStart = fullDateStr;
                    if (week === 2 && colAdded === (daysToDisplay - 1)) rangeEnd = fullDateStr;
                    
                    headerRow += `<th class="col-day">${currentDate.toLocaleDateString('en-US', {weekday:'short'}).toUpperCase()}<br><span style="font-weight:normal; font-size:12px;">${fullDateStr}</span></th>`;
                    colAdded++;
                    currentDate.setDate(currentDate.getDate() + 1);
                }
                thead.innerHTML = headerRow + '</tr>';

                // Data Rows Construction
                finalCategories.forEach(cat => {
                    let rowClass = cat.isRed ? 'class="non-veg-row"' : '';
                    let row = `<tr ${rowClass}><td class="col-category">${cat.name}</td>`;
                    for (let i = 0; i < daysToDisplay; i++) row += `<td><input type="text" class="cell-input" list="${cat.list}" value="${cat.def}"></td>`;
                    tbody.innerHTML += row + '</tr>';
                });

                table.appendChild(thead); table.appendChild(tbody);
                wrapper.appendChild(table);
                container.appendChild(wrapper);
            }
            document.getElementById('dateRangeDisplay').innerText = `LUNCH MENU: ${rangeStart} TO ${rangeEnd}`;
        }

        function resetMenu() {
            if (confirm("Are you sure you want to clear all menu items?")) {
                const inputs = document.querySelectorAll('.cell-input');
                inputs.forEach(input => input.value = "");
            }
        }

        function shareWithWhatsApp() {
            const loading = document.getElementById('loading');
            loading.style.display = 'block';

            setTimeout(async () => {
                try {
                    const element = document.getElementById('captureArea');
                    const canvas = await html2canvas(element, { 
                        scale: 4, 
                        useCORS: true,
                        backgroundColor: "#ffffff",
                        windowWidth: 1800 
                    });

                    canvas.toBlob(async (blob) => {
                        const file = new File([blob], 'DJ_Menu_Schedule.jpg', { type: 'image/jpeg' });

                        if (navigator.share && navigator.canShare({ files: [file] })) {
                            try {
                                await navigator.share({
                                    files: [file],
                                    title: 'Catering Menu',
                                    text: 'Here is the updated menu schedule.'
                                });
                            } catch (e) {
                                console.log("Share dismissed");
                            }
                        } else {
                            const link = document.createElement('a');
                            link.download = `DJ_Menu_Schedule.jpg`;
                            link.href = canvas.toDataURL("image/jpeg", 1.0);
                            link.click();
                            alert("High-Quality JPEG downloaded! Please attach it to WhatsApp manually.");
                        }
                        loading.style.display = 'none';
                    }, 'image/jpeg', 1.0); 
                    
                } catch (err) { 
                    alert("Error creating image: " + err); 
                    loading.style.display = 'none';
                }
            }, 800);
        }

        async function exportToExcel() {
            const clientNameRaw = document.getElementById('clientSelect').value;
            const dateRangeText = document.getElementById('dateRangeDisplay').innerText; // "LUNCH MENU: 10-02-2026 TO 23-02-2026"

            // --- FILENAME GENERATION ---
            // 1. Clean Client Name (lowercase, underscores)
            const safeClientName = clientNameRaw.toLowerCase().replace(/ /g, '_').replace(/[^a-z0-9_]/g, '');
            
            // 2. Extract Dates using Regex
            const dateMatches = dateRangeText.match(/(\d{2}-\d{2}-\d{4})/g);
            let dateSuffix = "";
            if (dateMatches && dateMatches.length >= 2) {
                dateSuffix = `_${dateMatches[0]}_${dateMatches[1]}`;
            }
            
            // 3. Final Filename
            const finalFileName = `Menu_${safeClientName}${dateSuffix}.xlsx`;


            // --- EXCEL GENERATION ---
            const workbook = new ExcelJS.Workbook();
            const worksheet = workbook.addWorksheet('Menu');

            // --- STYLES ---
            const navyFill = { type: 'pattern', pattern: 'solid', fgColor: { argb: 'FF1A3A5A' } }; // Brand Navy
            const redFill = { type: 'pattern', pattern: 'solid', fgColor: { argb: 'FFFFEBEE' } }; // Light Red (Non-veg bg)
            const grayFill = { type: 'pattern', pattern: 'solid', fgColor: { argb: 'FFF1F4F7' } }; // Gray (Category bg)
            
            const whiteFont = { name: 'Arial', color: { argb: 'FFFFFFFF' }, bold: true, size: 11 };
            const navyFont = { name: 'Arial', color: { argb: 'FF1A3A5A' }, bold: true, size: 10 };
            const redFont = { name: 'Arial', color: { argb: 'FFD32F2F' }, bold: true, size: 11 };
            const blackFont = { name: 'Arial', color: { argb: 'FF000000' }, size: 10 };

            const borderStyle = {
                top: { style: 'thin' },
                left: { style: 'thin' },
                bottom: { style: 'thin' },
                right: { style: 'thin' }
            };

            // --- HEADER INFO ---
            worksheet.addRow(["DJ HOSPITALITY & FACILITY MANAGEMENT PVT LTD"]);
            worksheet.getRow(1).font = { size: 16, bold: true, color: { argb: 'FF1A3A5A' } };
            worksheet.getRow(1).alignment = { horizontal: 'center' };
            worksheet.mergeCells('A1:H1');

            worksheet.addRow([`CLIENT: ${clientNameRaw}`]);
            worksheet.getRow(2).font = { size: 12, bold: true, color: { argb: 'FFA52A2A' } };
            worksheet.getRow(2).alignment = { horizontal: 'center' };
            worksheet.mergeCells('A2:H2');
            
            worksheet.addRow([dateRangeText]);
            worksheet.getRow(3).font = { size: 11, bold: true, color: { argb: 'FF555555' } };
            worksheet.getRow(3).alignment = { horizontal: 'center' };
            worksheet.mergeCells('A3:H3');

            worksheet.addRow([]); // Spacer

            // --- GET DATA FROM HTML ---
            const tables = document.querySelectorAll('table');
            const weekTitles = document.querySelectorAll('.week-title');

            tables.forEach((table, index) => {
                // 1. Add Week Title
                if(weekTitles[index]) {
                    const row = worksheet.addRow([weekTitles[index].innerText]);
                    row.font = whiteFont;
                    row.fill = navyFill;
                    row.alignment = { horizontal: 'center', vertical: 'middle' };
                    // Merge across estimated columns (Category + 7 Days = 8 cols)
                    worksheet.mergeCells(`A${row.number}:H${row.number}`);
                }

                // 2. Process Headers
                const headerRowData = [];
                table.querySelectorAll('th').forEach(th => {
                    headerRowData.push(th.innerText.replace(/\n/g, " ")); // Flatten multiline headers
                });
                
                const headerRow = worksheet.addRow(headerRowData);
                headerRow.height = 30;
                headerRow.eachCell((cell) => {
                    cell.fill = navyFill;
                    cell.font = whiteFont;
                    cell.alignment = { horizontal: 'center', vertical: 'middle', wrapText: true };
                    cell.border = borderStyle;
                });

                // 3. Process Rows
                const rows = table.querySelectorAll('tbody tr');
                rows.forEach(tr => {
                    const rowData = [];
                    // Category Name
                    rowData.push(tr.cells[0].innerText);
                    
                    // Input Values
                    for(let i = 1; i < tr.cells.length; i++) {
                        const input = tr.cells[i].querySelector('input');
                        rowData.push(input ? input.value : "");
                    }

                    const excelRow = worksheet.addRow(rowData);
                    excelRow.height = 25;

                    // Check if it's a Non-Veg row (by class or logic)
                    const isNonVeg = tr.classList.contains('non-veg-row');

                    excelRow.eachCell((cell, colNumber) => {
                        cell.border = borderStyle;
                        cell.alignment = { horizontal: 'center', vertical: 'middle' };

                        // Formatting for First Column (Category)
                        if (colNumber === 1) {
                            cell.alignment = { horizontal: 'left', vertical: 'middle', indent: 1 };
                            if (isNonVeg) {
                                cell.fill = redFill;
                                cell.font = redFont;
                            } else {
                                cell.fill = grayFill;
                                cell.font = navyFont;
                            }
                        } else {
                            // Formatting for Data Columns
                            if (isNonVeg) {
                                cell.font = redFont; // Red text for non-veg items
                            } else {
                                cell.font = blackFont;
                            }
                        }
                    });
                });

                worksheet.addRow([]); // Spacer between tables
            });

            // Set Column Widths
            worksheet.getColumn(1).width = 25; // Category
            for(let i=2; i<=9; i++) worksheet.getColumn(i).width = 20; // Days

            // Export with the new Filename
            const buffer = await workbook.xlsx.writeBuffer();
            saveAs(new Blob([buffer]), finalFileName);
        }
