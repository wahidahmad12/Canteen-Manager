    <script>
        let selectedRowForTransfer = null;

        document.getElementById('a-date').valueAsDate = new Date();

        function toggleSelectAll(masterCb) {
            const checkboxes = document.querySelectorAll('.asset-cb');
            checkboxes.forEach(cb => cb.checked = masterCb.checked);
        }

        function promptAddNewLocation() {
            const newLoc = prompt("Enter new Canteen Location name:");
            if (newLoc && newLoc.trim() !== "") {
                const trimmedLoc = newLoc.trim();
                const dropdowns = document.querySelectorAll('.location-dropdown');
                dropdowns.forEach(select => {
                    let exists = Array.from(select.options).some(opt => opt.value.toLowerCase() === trimmedLoc.toLowerCase());
                    if (!exists) {
                        const opt = document.createElement('option');
                        opt.value = trimmedLoc;
                        opt.textContent = trimmedLoc;
                        select.appendChild(opt);
                    }
                    select.value = trimmedLoc;
                });
                alert(`Location "${trimmedLoc}" added!`);
            }
        }

        function addAssetEntry(e) {
            e.preventDefault();
            const id = document.getElementById('a-id').value;
            const date = document.getElementById('a-date').value;
            const name = document.getElementById('a-name').value;
            const vendor = document.getElementById('a-vendor').value;
            const loc = document.getElementById('a-loc').value;
            const cost = document.getElementById('a-cost').value;
            const dep = document.getElementById('a-dep').value;
            const status = document.getElementById('a-status').value;

            let badgeClass = 'badge-active';
            if(status === 'Under Maintenance') badgeClass = 'badge-maintenance';
            if(status === 'Retired') badgeClass = 'badge-retired';

            const tbody = document.getElementById('asset-table-body');
            const row = document.createElement('tr');
            row.innerHTML = `

            selectedRowForTransfer = btn.closest('tr');

            document.getElementById('modal-asset-info').innerText = `Asset: ${assetId} - ${assetName} (Current: ${currentLoc})`;
            document.getElementById('transferModal').style.display = 'flex';

            document.getElementById('transferModal').style.display = 'none';

            }
        }

                return;
            }

            const items = Array.from(selectedRows).map(cb => extractRowData(cb.closest('tr')));
            renderAndPrintTags(items);
        }

        function extractRowData(row) {
            return {
                id: row.querySelector('.asset-id-cell').innerText,
                name: row.querySelector('.asset-name-cell').innerText,
                date: row.querySelector('.asset-date-cell').innerText,
                loc: row.querySelector('.loc-text').innerText
            };
        }

        function renderAndPrintTags(items) {
            const container = document.getElementById('printable-batch-container');
            container.innerHTML = '';

            items.forEach((item, idx) => {
                const { tagDiv, qrId } = createTagElement(item.id, item.name, item.date, item.loc, idx);
                container.appendChild(tagDiv);

                const qrData = `Asset ID: ${item.id}\nName: ${item.name}\nDate: ${item.date}\nLoc: ${item.loc}\nOwner: DJ Hospitality`;
                new QRCode(document.getElementById(qrId), {
                    text: qrData,
                    width: 96,
                    height: 96,
                    colorDark: "#000000",
                    colorLight: "#ffffff",
                    correctLevel: QRCode.CorrectLevel.H
                });
            });

            setTimeout(() => {
                window.print();
            }, 150);
        }
    </script>
